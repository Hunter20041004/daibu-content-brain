# 四套公司中立輪播模板原型 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `executing-plans` to implement this plan task-by-task. Do not dispatch subagents unless the user explicitly requests subagent execution.

**Goal:** 建立四套彼此獨立、各七頁、只替換公司 Logo 與文字即可使用的 1080×1350 輪播視覺原型，讓使用者逐套與同頁並排比較。

**Architecture:** 原型完全位於 `.superpowers/prototypes/company-neutral-carousel/`，由 Python build script 將既有三家公司 stress fixture 匯出成內嵌 JSON，再生成單一使用者介面 `index.html`。HTML 以 Tailwind CDN 負責外層比較器，貼文畫布使用檔頭 CSS design tokens 與四套獨立 page renderer；Python `unittest` 負責靜態契約，Playwright Python 負責 168 個實際瀏覽器狀態與互動契約。

**Tech Stack:** Python 3.11、標準函式庫 `unittest`／`http.server`、既有 `.venv` Playwright Chromium、單檔 HTML、Tailwind CDN、CSS variables、既有 SVG 呆一布與固定 AI 生成原型資產。

## Global Constraints

- 固定畫布為 `1080×1350`、比例 `4:5`；瀏覽器只縮放畫布，不重排畫布內部內容。
- 四套風格各有完整七頁，共 28 個主要頁面；不得用同一座標骨架只換皮。
- 日後只替換公司 Logo 與七頁文字即可使用；不得為個別公司生成插畫、背景或專屬 CSS。
- SUN、Billow AI Labs、Shiraz AI 只作壓力測試；固定 HTML、CSS、SVG、PNG 不得烘焙三家公司名稱、Logo 或產業物件。
- 不修改 `daibu/cards/templates/`、`daibu/cards/layouts/`、SQLite、正式模板版本或現行 localhost 服務。
- 現有工作樹有使用者未提交的 `docs/2026-08-07-呆一布內容工作站是怎麼做出來的.html`、`docs/build-story/template.html` 等變更；不得暫存、格式化、還原或提交它們。
- 嚴格垂直 TDD：一次只增加一個失敗契約、一個頁面最小實作，再轉綠；禁止先寫完七頁後補測試。
- 原型比較器是無 build 流程的 HTML 工具：使用 Tailwind CDN，並在 `<style>` 首段定義所有色彩、字級、間距、圓角與陰影 design tokens。
- 基礎控制使用原生 `button`，保留 focus、selected、disabled 語意；不手刻假按鈕 div。
- 視覺 QA 至少兩輪；每輪只保存 1280×800 與 375×812 各一張截圖。
- 奇幻與 3D 圖像只作固定模板資產；生成一次後同樣套在三家公司，不依公司或內容變化。

```python
from __future__ import annotations

from dataclasses import asdict
import json
from pathlib import Path
import sys

ROOT = Path(__file__).resolve().parent
PROJECT_ROOT = ROOT.parents[2]
if str(PROJECT_ROOT) not in sys.path:
    sys.path.insert(0, str(PROJECT_ROOT))

from tests.card_fixtures import company_stress_draft

COMPANIES = ("SUN", "Billow AI Labs", "Shiraz AI")
LOGOS = {
    "SUN": "/daibu/cards/stress_assets/logo-wide.svg",
    "Billow AI Labs": "/daibu/cards/stress_assets/logo-tall.svg",
    "Shiraz AI": "/.superpowers/prototypes/company-neutral-carousel/assets/logo-square.svg",
}


def company_payload() -> list[dict[str, object]]:
    payload = []
    for company in COMPANIES:
        draft = company_stress_draft(company)
        pages = []
        for page in draft.pages:
            pages.append({
                "page_no": page.page_no,
                "page_type": page.page_type,
                "heading": page.heading,
                "sections": {
                    section.label: section.text for section in page.sections
                },
                "diagram": [] if page.diagram is None else [
                    asdict(node) for node in page.diagram.nodes
                ],
            })
        payload.append({"company": company, "logo": LOGOS[company], "pages": pages})
    return payload


def build() -> Path:
    template = (ROOT / "template.html").read_text(encoding="utf-8")
    serialized = json.dumps(company_payload(), ensure_ascii=False).replace("</", "<\\/")
    if "__COMPANY_FIXTURES__" not in template:
        raise RuntimeError("template missing __COMPANY_FIXTURES__ marker")
    output = template.replace("__COMPANY_FIXTURES__", serialized)
    target = ROOT / "index.html"
    target.write_text(output, encoding="utf-8")
    return target


if __name__ == "__main__":
    print(build())
```

```html
<script>
  const STYLES = [
    ["notebook", "粉彩手帳"],
    ["y2k", "Y2K 科技拼貼"],
    ["fantasy", "奇幻世界觀"],
    ["editorial", "紙感 3D 資訊圖解"],
  ];
  const state = { style: "notebook", company: "SUN", page: 1, mode: "single" };
  const fixtures = Object.fromEntries(window.COMPANY_FIXTURES.map(item => [item.company, item]));
  const controls = document.querySelector(".prototype-controls");
  const stage = document.querySelector(".prototype-stage");

  function button(group, value, label) {
    const selected = state[group] === value;
    return `<button type="button" data-control="${group}" data-value="${value}" aria-pressed="${selected}">${label}</button>`;
  }

  function renderControls() {
    controls.innerHTML = `
      <div class="control-row" aria-label="風格">${STYLES.map(([v,l]) => button("style", v, l)).join("")}</div>
      <div class="control-row" aria-label="公司">${Object.keys(fixtures).map(v => button("company", v, v)).join("")}</div>
      <div class="control-row" aria-label="頁面">${Array.from({length: 7}, (_,i) => button("page", i+1, `第 ${i+1} 頁`)).join("")}</div>
      <div class="control-row" aria-label="模式">${button("mode", "single", "逐套查看")}${button("mode", "compare", "同頁比較")}</div>`;
  }

  function pageData(company, page) { return fixtures[company].pages[page - 1]; }
  function field(page, label) { return page.sections[label] || ""; }
  function renderCard(style, company, pageNo) {
    const page = pageData(company, pageNo);
    return `<article class="post style-${style} page-${pageNo}" data-style="${style}" data-company="${company}" data-page="${pageNo}"><p>${page.heading}</p></article>`;
  }
  function renderStage() {
    const styles = state.mode === "compare" ? STYLES.map(([value]) => value) : [state.style];
    stage.innerHTML = styles.map(style => `<div class="post-frame" data-frame-style="${style}">${renderCard(style, state.company, state.page)}</div>`).join("");
    requestAnimationFrame(scaleFrames);
  }
  function scaleFrames() {
    document.querySelectorAll(".post-frame").forEach(frame => {
      const post = frame.querySelector(".post");
      const scale = frame.clientWidth / 1080;
      post.style.transform = `scale(${scale})`;
    });
  }
  function setState(next) {
    Object.assign(state, next);
    renderControls();
    renderStage();
  }
  controls.addEventListener("click", event => {
    const target = event.target.closest("button[data-control]");
    if (!target) return;
    const group = target.dataset.control;
    const raw = target.dataset.value;
    setState({ [group]: group === "page" ? Number(raw) : raw });
  });
  window.addEventListener("resize", scaleFrames);
  window.__prototype = { setState, renderCard, getState: () => ({...state}) };
  setState(state);
</script>
```

from functools import partial
from http.server import SimpleHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from threading import Thread
import unittest

from playwright.sync_api import sync_playwright

ROOT = Path(__file__).resolve().parent
PROJECT_ROOT = ROOT.parents[2]
COMPANIES = ("SUN", "Billow AI Labs", "Shiraz AI")


class QuietHandler(SimpleHTTPRequestHandler):
    def log_message(self, format, *args):
        pass


def start_server(root: Path):
    handler = partial(QuietHandler, directory=str(root))
    server = ThreadingHTTPServer(("127.0.0.1", 0), handler)
    thread = Thread(target=server.serve_forever, daemon=True)
    thread.start()
    return server, server.server_address[1]


class BrowserContract(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.server, cls.port = start_server(PROJECT_ROOT)
        cls.playwright = sync_playwright().start()
        cls.browser = cls.playwright.chromium.launch()
        cls.page = cls.browser.new_page(viewport={"width": 1280, "height": 800})
        cls.page.goto(f"http://127.0.0.1:{cls.port}/.superpowers/prototypes/company-neutral-carousel/index.html")

    @classmethod
    def tearDownClass(cls):
        cls.page.close()
        cls.browser.close()
        cls.playwright.stop()
        cls.server.shutdown()
        cls.server.server_close()

    def test_controls_switch_style_company_page_and_compare_mode(self):
        api = self.page.evaluate("() => Object.keys(window.__prototype)")
        self.assertEqual(sorted(api), ["getState", "renderCard", "setState"])
        self.page.get_by_role("button", name="Billow AI Labs").click()
        self.page.get_by_role("button", name="第 4 頁").click()
        self.page.get_by_role("button", name="同頁比較").click()
        self.assertEqual(self.page.locator(".post").count(), 4)
        self.assertEqual(self.page.locator(".post[data-company='Billow AI Labs'][data-page='4']").count(), 4)


if __name__ == "__main__":
    unittest.main()
